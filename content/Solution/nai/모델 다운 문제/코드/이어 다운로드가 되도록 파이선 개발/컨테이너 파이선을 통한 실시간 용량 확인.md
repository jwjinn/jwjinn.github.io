## 다운로드 하고 있는 POD이름 확인 후 모니터링

```shell

# 1. 파드 이름 확인
POD_NAME=$(kubectl get pods -n nai-admin -l app=resume-downloader -o jsonpath="{.items[0].metadata.name}")

# 2. 5초마다 파일 용량 확인 명령어 반복 실행
while true; do
  echo "--- Checking File Size ---"
  kubectl exec -n nai-admin $POD_NAME -- du -sh /data/model-files
  sleep 10
done

```
