flowchart TD
    %% --- 스타일 정의 ---
    classDef crd fill:#E8EAF6,stroke:#1A237E,stroke-width:2px,color:#1A237E,stroke-dasharray: 5 5
    classDef config fill:#E1F5FE,stroke:#0277BD,stroke-width:2px,color:#01579B
    classDef workload fill:#F3E5F5,stroke:#7B1FA2,stroke-width:2px,color:#4A148C

    %% --- 1. 최상위 리소스 (Model) ---
    %% 따옴표로 감싸서 특수문자 오류 방지
    Model[("Model CRD<br>(nai-96141...-cd)<br>Spec: Qwen3 / 276G")]:::crd

    %% --- 2. 설정 및 스토리지 리소스 ---
    SC{{"StorageClass<br>(nai-nfs-storage)"}}:::config
    PVC["Persistent Volume Claim<br>(nai-...-pvc-claim)"]:::config

    %% --- 연결 흐름 (Orchestration) ---

    %% --- 기존 연결 흐름 ---
    PVC -.->|"Bound"| SC
    SC ==>|"Provisioning"| NFS
    
    Job -->|"Manage"| Pod
    Pod -->|"Volume Mount"| PVC
    Pod == "데이터 다운로드/저장" ==> NFS

    %% --- 그래프 스타일 ---
    linkStyle default stroke:#666,stroke-width:1px,fill:none
